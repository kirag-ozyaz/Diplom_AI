& {
    $ErrorActionPreference = "Stop"
    $src = "V2/data/chunked_merged_manual"
    $dst = "V2/data/chunked_active_manual"
    if (Test-Path $dst) { throw "Папка уже существует: $dst" }
    New-Item -ItemType Directory -Path $dst | Out-Null
    Copy-Item "$src/*.jsonl" -Destination $dst
    Copy-Item "$src/assets" -Destination "$dst/assets" -Recurse
    (Get-ChildItem $dst -Filter *.jsonl).Count
    (Get-ChildItem $dst -Filter *.jsonl | Get-Content | Measure-Object -Line).Lines

Создаём отдельный конфиг для нового корпуса — основной `pipeline.json` не меняем. Выполните из корня проекта:
& {
    $ErrorActionPreference = "Stop"
    $newConfig = "V2/configs/pipeline_manual.json"
    if (Test-Path $newConfig) { throw "Файл уже существует: $newConfig" }
    $config = Get-Content "V2/configs/pipeline.json" -Raw -Encoding UTF8 | ConvertFrom-Json
    $config.paths.chunks = "V2/data/chunked_active_manual"
    $config | ConvertTo-Json -Depth 50 | Set-Content $newConfig -Encoding UTF8
    python V2/scripts/run_pipeline.py validate --corpus-only --config V2/configs/pipeline_manual.json
}
